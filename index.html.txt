const SoundFX = {
  ctx: null,
  muted: false,
  bgmInterval: null,
  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === "suspended") {
      this.ctx.resume().catch(() => {});
    }
  },
  play(type) {
    if (this.muted) return;
    this.init();
    if (!this.ctx) return;
    const now = this.ctx.currentTime;

    if (type === "click") {
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = "triangle";
      osc.frequency.setValueAtTime(520, now);
      osc.frequency.exponentialRampToValueAtTime(180, now + 0.055);
      gain.gain.setValueAtTime(0.14, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.06);
      osc.connect(gain); gain.connect(this.ctx.destination);
      osc.start(now); osc.stop(now + 0.06);
    } else if (type === "cash") {
      [880, 1175, 1568].forEach((freq, idx) => {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(freq, now + idx * 0.06);
        gain.gain.setValueAtTime(0.15, now + idx * 0.06);
        gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.06 + 0.18);
        osc.connect(gain); gain.connect(this.ctx.destination);
        osc.start(now + idx * 0.06); osc.stop(now + idx * 0.06 + 0.19);
      });
    } else if (type === "build") {
      [140, 110].forEach((freq, idx) => {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = "square";
        osc.frequency.setValueAtTime(freq, now + idx * 0.12);
        osc.frequency.exponentialRampToValueAtTime(55, now + idx * 0.12 + 0.09);
        gain.gain.setValueAtTime(0.16, now + idx * 0.12);
        gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.12 + 0.1);
        osc.connect(gain); gain.connect(this.ctx.destination);
        osc.start(now + idx * 0.12); osc.stop(now + idx * 0.12 + 0.1);
      });
    } else if (type === "levelup") {
      [440, 554, 659, 880].forEach((freq, i) => {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = "triangle";
        osc.frequency.setValueAtTime(freq, now + i * 0.09);
        gain.gain.setValueAtTime(0.18, now + i * 0.09);
        gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.09 + 0.3);
        osc.connect(gain); gain.connect(this.ctx.destination);
        osc.start(now + i * 0.09); osc.stop(now + i * 0.09 + 0.31);
      });
    } else if (type === "pistol" || type === "rifle") {
      const bufferSize = this.ctx.sampleRate * 0.08;
      const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.28));
      const noise = this.ctx.createBufferSource();
      noise.buffer = buffer;
      const filter = this.ctx.createBiquadFilter();
      filter.type = "bandpass";
      filter.frequency.value = type === "pistol" ? 1300 : 950;
      const gain = this.ctx.createGain();
      gain.gain.setValueAtTime(0.16, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
      noise.connect(filter); filter.connect(gain); gain.connect(this.ctx.destination);
      noise.start(now);
    } else if (type === "explosion") {
      const bufferSize = this.ctx.sampleRate * 0.35;
      const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.32));
      const noise = this.ctx.createBufferSource();
      noise.buffer = buffer;
      const filter = this.ctx.createBiquadFilter();
      filter.type = "lowpass";
      filter.frequency.setValueAtTime(320, now);
      filter.frequency.exponentialRampToValueAtTime(60, now + 0.34);
      const gain = this.ctx.createGain();
      gain.gain.setValueAtTime(0.28, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);
      noise.connect(filter); filter.connect(gain); gain.connect(this.ctx.destination);
      noise.start(now);
    } else if (type === "siren") {
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = "sawtooth";
      osc.frequency.setValueAtTime(620, now);
      osc.frequency.linearRampToValueAtTime(920, now + 0.25);
      osc.frequency.linearRampToValueAtTime(620, now + 0.5);
      gain.gain.setValueAtTime(0.12, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.52);
      osc.connect(gain); gain.connect(this.ctx.destination);
      osc.start(now); osc.stop(now + 0.53);
    }
  },
  startAmbient() {
    if (this.bgmInterval) return;
    let step = 0;
    const bassNotes = [73.42, 73.42, 87.31, 65.41];
    this.bgmInterval = setInterval(() => {
      if (this.muted || !this.ctx || this.ctx.state !== "running") return;
      const now = this.ctx.currentTime;
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = "triangle";
      osc.frequency.setValueAtTime(bassNotes[step % bassNotes.length], now);
      gain.gain.setValueAtTime(0.04, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.45);
      osc.connect(gain); gain.connect(this.ctx.destination);
      osc.start(now); osc.stop(now + 0.46);
      step++;
    }, 650);
  }
};

// ================= PERSISTENCIA DE PARTIDA (GUARDADO AUTOMÁTICO) =================
const SAVE_KEY = "cw3d_save_state_v3";

function saveGameProgress() {
  if (!player.name) return;
  const state = {
    player,
    myBuildings: myBuildings.map(b => ({
      id: b.id, type: b.type, x: b.x, z: b.z, lvl: b.lvl,
      stored: b.stored, burning: !!b.burning, destroyed: !!b.destroyed,
      hp: b.hp || getBuildingMaxHp(b)
    })),
    activeTimers: activeTimers.map(t => ({
      id: t.id, label: t.label, endTime: t.endTime,
      actionType: t.actionType, payload: t.payload
    })),
    lastSaved: Date.now()
  };
  try {
    localStorage.setItem(SAVE_KEY, JSON.stringify(state));
  } catch (e) { console.warn("Error guardando progreso local", e); }

  if (ADMIN_CONFIG.firebaseDbUrl) {
    const syncData = {
      id: player.id, name: player.name, level: player.level,
      lootCash: Math.max(300, Math.floor(player.cash * 0.4)),
      lootGoods: Math.max(120, Math.floor(player.goods * 0.4)),
      pendingEvent: null,
      layout: myBuildings.filter(b => !b.destroyed).map(b => ({
        type: b.type, x: b.x, z: b.z, hp: getBuildingMaxHp(b)
      }))
    };
    fetch(`${ADMIN_CONFIG.firebaseDbUrl}/players/${player.id}.json`, {
      method: "PUT", body: JSON.stringify(syncData)
    }).catch(() => {});
  }
}

function hydrateTimerCallback(t) {
  if (t.actionType === "build") {
    return () => {
      const spec = BUILDINGS_DB[t.payload.type];
      myBuildings.push({
        id: Date.now(), type: t.payload.type, x: t.payload.x, z: t.payload.z,
        lvl: 1, stored: 0, burning: false, destroyed: false, hp: spec.baseHp
      });
      addXP(160); rebuildBase3D(); updateHUD(); saveGameProgress();
      SoundFX.play("build");
      showToast(`¡${spec.name} construida!`);
    };
  } else if (t.actionType === "upgrade") {
    return () => {
      const b = myBuildings.find(item => item.id === t.payload.bId);
      if (b) {
        b.lvl++;
        b.destroyed = false;
        b.burning = false;
        b.hp = getBuildingMaxHp(b);
        addXP(240 * b.lvl); rebuildBase3D(); updateHUD(); saveGameProgress();
        SoundFX.play("build");
        showToast(`¡${BUILDINGS_DB[b.type].name} subió a Nivel ${b.lvl}!`);
      }
    };
  } else if (t.actionType === "train") {
    return () => {
      const spec = TROOPS_DB[t.payload.tKey];
      player.troops[t.payload.tKey]++;
      addXP(65); rebuildBase3D(); updateHUD(); saveGameProgress();
      SoundFX.play("click");
      showToast(`¡${spec.name} listo para el combate!`);
    };
  }
  return () => {};
}

function loadSavedGame() {
  try {
    const raw = localStorage.getItem(SAVE_KEY);
    if (!raw) return false;
    const saved = JSON.parse(raw);
    if (!saved || !saved.player || !saved.player.name) return false;

    player = { ...player, ...saved.player };
    if (Array.isArray(saved.myBuildings) && saved.myBuildings.length > 0) {
      myBuildings = saved.myBuildings.map(b => ({
        ...b,
        destroyed: !!b.destroyed,
        burning: !!b.burning,
        hp: b.destroyed ? 0 : (b.hp || getBuildingMaxHp(b))
      }));
    }

    if (saved.lastSaved) {
      const elapsedSec = Math.min(14400, Math.max(0, (Date.now() - saved.lastSaved) / 1000));
      myBuildings.forEach(b => {
        if (b.burning || b.destroyed) return;
        const s = BUILDINGS_DB[b.type];
        if (s && s.prod) {
          b.stored += s.rate * (1 + (b.lvl - 1) * 0.35) * 0.25 * elapsedSec;
        }
      });
    }

    activeTimers = [];
    if (Array.isArray(saved.activeTimers)) {
      saved.activeTimers.forEach(t => {
        const cb = hydrateTimerCallback(t);
        if (Date.now() >= t.endTime) {
          cb();
        } else {
          activeTimers.push({ ...t, onComplete: cb });
        }
      });
    }

    document.getElementById("login-screen").style.display = "none";
    rebuildBase3D();
    updateHUD();
    showToast(`Bienvenido de vuelta, ${player.name}`);
    return true;
  } catch (e) {
    console.warn("No se pudo cargar partida previa", e);
    return false;
  }
}

async function registerOrLoginPlayer() {
  SoundFX.init(); SoundFX.play("click");
  const rawName = document.getElementById("username-input").value.trim();
  const errBox = document.getElementById("login-error");

  if (rawName.length < 3) {
    errBox.textContent = "El nombre debe tener al menos 3 caracteres.";
    return;
  }
  if (!/^[a-zA-Z0-9_áéíóúÁÉÍÓÚñÑ-]+$/.test(rawName)) {
    errBox.textContent = "Usa solo letras, números o guiones.";
    return;
  }

  const normalized = rawName.toLowerCase();
  const localRegistry = getSavedUsersRegistry();
  const takenInDirectory = onlinePlayersDirectory.some(u => u.name.toLowerCase() === normalized);
  const takenLocally = localRegistry.some(u => u.name.toLowerCase() === normalized);

  let takenInFirebase = false;
  if (ADMIN_CONFIG.firebaseDbUrl) {
    try {
      const res = await fetch(`${ADMIN_CONFIG.firebaseDbUrl}/players.json`);
      const data = await res.json();
      if (data) {
        onlinePlayersDirectory = Object.values(data);
        takenInFirebase = onlinePlayersDirectory.some(u => u.name.toLowerCase() === normalized);
      }
    } catch (e) { console.warn("Sincronizando en modo local"); }
  }

  if (takenInDirectory || takenLocally || takenInFirebase) {
    errBox.textContent = `El nombre "${rawName}" ya está en uso. Elige otro.`;
    return;
  }

  player.name = rawName;
  localRegistry.push({ id: player.id, name: player.name });
  localStorage.setItem("cw3d_users_registry", JSON.stringify(localRegistry));

  onlinePlayersDirectory.push({
    id: player.id, name: player.name, level: player.level,
    lootCash: player.cash, lootGoods: player.goods, pendingEvent: null,
    layout: myBuildings.map(b => ({ type: b.type, x: b.x, z: b.z, hp: BUILDINGS_DB[b.type].baseHp }))
  });

  saveGameProgress();
  document.getElementById("login-screen").style.display = "none";
  updateHUD();
  SoundFX.play("levelup");
  showToast(`Hacienda fundada por ${player.name}`);
}

// ================= PROGRESIÓN Y ECONOMÍA =================
function xpForNextLevel(lvl) { return Math.floor(600 * Math.pow(lvl, 1.65)); }
function getMaxTroops() { return 3 + player.level; }
function getCurrentRankName() {
  const idx = Math.min(9, Math.floor((player.level - 1) / 10));
  return `Rango ${idx + 1}: ${RANKS[idx].split(" ")[0]}`;
}
function addXP(baseAmount) {
  player.xp += Math.floor(baseAmount * ADMIN_CONFIG.xpMultiplier);
  let req = xpForNextLevel(player.level);
  let leveled = false;
  while (player.xp >= req && player.level < 100) {
    player.xp -= req; player.level++; req = xpForNextLevel(player.level);
    leveled = true;
    showToast(`¡SUBISTE A NIVEL ${player.level}! (${getCurrentRankName()})`);
  }
  if (leveled) SoundFX.play("levelup");
  updateHUD();
  saveGameProgress();
}

function createHDTextureSet(type) {
  const size = 1024;
  const colorCanvas = document.createElement("canvas");
  const bumpCanvas = document.createElement("canvas");
  colorCanvas.width = bumpCanvas.width = size;
  colorCanvas.height = bumpCanvas.height = size;
  const ctx = colorCanvas.getContext("2d");
  const btx = bumpCanvas.getContext("2d");

  if (type === "terrain") {
    ctx.fillStyle = "#243f1b"; ctx.fillRect(0, 0, size, size);
    btx.fillStyle = "#808080"; btx.fillRect(0, 0, size, size);
    for (let i = 0; i < 55000; i++) {
      const x = Math.random() * size, y = Math.random() * size, r = 1 + Math.random() * 4;
      const shades = ["#1a3013", "#2e4e22", "#3b5e29", "#4b3d26", "#152610"];
      ctx.fillStyle = shades[Math.floor(Math.random() * shades.length)];
      ctx.globalAlpha = 0.42; ctx.fillRect(x, y, r, r);
      btx.fillStyle = Math.random() > 0.5 ? "#acacac" : "#545454";
      btx.globalAlpha = 0.34; btx.fillRect(x, y, r, r);
    }
    ctx.globalAlpha = 0.38; ctx.strokeStyle = "#5c442c"; ctx.lineWidth = 48; ctx.lineCap = "round";
    ctx.beginPath(); ctx.moveTo(512, 1024); ctx.quadraticCurveTo(475, 640, 512, 420); ctx.stroke();
  } else if (type === "stucco") {
    ctx.fillStyle = "#e6d4b5"; ctx.fillRect(0, 0, size, size);
    btx.fillStyle = "#808080"; btx.fillRect(0, 0, size, size);
    for (let i = 0; i < 28000; i++) {
      const x = Math.random() * size, y = Math.random() * size;
      ctx.fillStyle = Math.random() > 0.5 ? "#cdb896" : "#f5ebd8";
      ctx.globalAlpha = 0.32; ctx.fillRect(x, y, 4, 4);
      btx.fillStyle = Math.random() > 0.5 ? "#b8b8b8" : "#484848";
      btx.globalAlpha = 0.38; btx.fillRect(x, y, 4, 4);
    }
  } else if (type === "roof") {
    ctx.fillStyle = "#8f2f1b"; ctx.fillRect(0, 0, size, size);
    btx.fillStyle = "#383838"; btx.fillRect(0, 0, size, size);
    for (let y = 0; y < size; y += 36) {
      for (let x = 0; x < size; x += 48) {
        const grad = ctx.createLinearGradient(x, y, x + 48, y);
        grad.addColorStop(0, "#631c0d"); grad.addColorStop(0.5, "#b54228"); grad.addColorStop(1, "#541609");
        ctx.fillStyle = grad; ctx.fillRect(x, y, 46, 34);
        btx.fillStyle = "#e0e0e0"; btx.fillRect(x + 4, y + 2, 38, 30);
      }
    }
  } else if (type === "soil") {
    ctx.fillStyle = "#301d11"; ctx.fillRect(0, 0, size, size);
    btx.fillStyle = "#404040"; btx.fillRect(0, 0, size, size);
    for (let y = 16; y < size; y += 56) {
      ctx.fillStyle = "#1f1008"; ctx.fillRect(0, y, size, 22);
      ctx.fillStyle = "#472c1b"; ctx.fillRect(0, y + 22, size, 24);
      btx.fillStyle = "#151515"; btx.fillRect(0, y, size, 22);
      btx.fillStyle = "#e6e6e6"; btx.fillRect(0, y + 22, size, 24);
    }
  } else if (type === "concrete") {
    ctx.fillStyle = "#5c6163"; ctx.fillRect(0, 0, size, size);
    btx.fillStyle = "#808080"; btx.fillRect(0, 0, size, size);
    for (let i = 0; i < 24000; i++) {
      const x = Math.random() * size, y = Math.random() * size;
      ctx.fillStyle = Math.random() > 0.5 ? "#484c4e" : "#757a7d";
      ctx.globalAlpha = 0.3; ctx.fillRect(x, y, 3, 3);
      btx.fillStyle = Math.random() > 0.5 ? "#a2a2a2" : "#5e5e5e";
      btx.globalAlpha = 0.34; btx.fillRect(x, y, 3, 3);
    }
    ctx.globalAlpha = 0.65; ctx.strokeStyle = "#26292a"; ctx.lineWidth = 5;
    btx.globalAlpha = 0.92; btx.strokeStyle = "#111111"; btx.lineWidth = 6;
    for (let g = 0; g < size; g += 256) { ctx.strokeRect(g, 0, 256, size); btx.strokeRect(g, 0, 256, size); }
  }

  const map = new THREE.CanvasTexture(colorCanvas);
  const bumpMap = new THREE.CanvasTexture(bumpCanvas);
  map.wrapS = map.wrapT = bumpMap.wrapS = bumpMap.wrapT = THREE.RepeatWrapping;
  return { map, bumpMap };
}

function populateJungle() {
  const trunkMat = new THREE.MeshStandardMaterial({ color: 0x473222, roughness: 0.9 });
  const palmMat = new THREE.MeshStandardMaterial({ color: 0x245420, roughness: 0.72, side: THREE.DoubleSide });
  const bushMat = new THREE.MeshStandardMaterial({ color: 0x1c421a, roughness: 0.85 });
  for (let i = 0; i < 160; i++) {
    const ang = Math.random() * Math.PI * 2, dist = 33 + Math.random() * 72;
    const x = Math.cos(ang) * dist, z = Math.sin(ang) * dist;
    if (i % 2 === 0) {

      const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.35, h, 8), trunkMat);

        canopy.castShadow = true; tree.add(canopy);
      }

    }
  }
}

function createHealthBarSprite(widthWorld = 3.2, heightWorld = 0.42) {
  const canvas = document.createElement("canvas");
  canvas.width = 256;
  canvas.height = 36;
  const ctx = canvas.getContext("2d");
  const tex = new THREE.CanvasTexture(canvas);
  const mat = new THREE.SpriteMaterial({ map: tex, transparent: true, depthTest: false });
  const sprite = new THREE.Sprite(mat);
  sprite.scale.set(widthWorld, heightWorld, 1);
  sprite.userData = { canvas, ctx, tex, lastPct: -1, lastSelected: false };
  updateHealthBarSprite(sprite, 1, 1, false);
  return sprite;
}

function updateHealthBarSprite(sprite, hp, maxHp, isSelected = false) {
  if (!sprite || !sprite.userData) return;
  const pct = Math.max(0, Math.min(1, hp / (maxHp || 1)));
  if (Math.abs(sprite.userData.lastPct - pct) < 0.005 && sprite.userData.lastSelected === isSelected) return;
  sprite.userData.lastPct = pct;
  sprite.userData.lastSelected = isSelected;

  const { canvas, ctx, tex } = sprite.userData;
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  ctx.fillStyle = "rgba(12, 10, 8, 0.88)";
  ctx.fillRect(4, 4, canvas.width - 8, canvas.height - 8);

  ctx.strokeStyle = isSelected ? "#ff3b30" : "#d4af37";
  ctx.lineWidth = isSelected ? 5 : 3;
  ctx.strokeRect(4, 4, canvas.width - 8, canvas.height - 8);

  let barColor = "#2ecc71";
  if (pct < 0.35) barColor = "#e74c3c";
  else if (pct < 0.65) barColor = "#f1c40f";

  ctx.fillStyle = barColor;
  ctx.fillRect(9, 9, (canvas.width - 18) * pct, canvas.height - 18);

  tex.needsUpdate = true;
}

function createRuinsMesh3D(b) {
  const group = new THREE.Group();
  group.position.set(b.x, 0, b.z);
  const rad = BUILDINGS_DB[b.type].radius;

  const scorchMat = new THREE.MeshStandardMaterial({ color: 0x1c1917, roughness: 0.96 });
  const rubbleMat = new THREE.MeshStandardMaterial({ color: 0x44403c, bumpMap: matConcreteSet.bumpMap, bumpScale: 0.2, roughness: 0.92 });
  const beamMat = new THREE.MeshStandardMaterial({ color: 0x291d14, roughness: 0.9 });

  // Base carbonizada
  const crater = new THREE.Mesh(new THREE.CylinderGeometry(rad * 0.88, rad * 0.98, 0.22, 16), scorchMat);
  crater.position.y = 0.11;
  crater.receiveShadow = true;
  group.add(crater);

  // Bloques de concreto derrumbados
  for (let i = 0; i < 11; i++) {
    const ang = (i / 11) * Math.PI * 2 + 0.3;
    const dist = rad * (0.25 + (i % 3) * 0.22);
    const block = new THREE.Mesh(
      new THREE.DodecahedronGeometry(0.45 + (i % 4) * 0.16, 0),
      rubbleMat
    );
    block.position.set(Math.cos(ang) * dist, 0.38, Math.sin(ang) * dist);
    block.rotation.set(i * 0.5, i * 0.8, i * 0.3);
    block.castShadow = true;
    group.add(block);
  }

  // Vigas quemadas inclinadas
  for (let v = 0; v < 3; v++) {
    const beam = new THREE.Mesh(new THREE.BoxGeometry(0.22, 2.1, 0.22), beamMat);
    beam.position.set((v - 1) * 0.85, 0.8, (v % 2 === 0 ? 0.5 : -0.5));
    beam.rotation.set(0.45 * (v - 1), 0.3, 0.55 - v * 0.4);
    beam.castShadow = true;
    group.add(beam);
  }

  // Anillo rojo de alerta en el suelo
  const alertRing = new THREE.Mesh(
    new THREE.RingGeometry(rad * 0.86, rad * 1.0, 28),
    new THREE.MeshBasicMaterial({ color: 0xe74c3c, side: THREE.DoubleSide, transparent: true, opacity: 0.72 })
  );
  alertRing.rotation.x = -Math.PI / 2;
  alertRing.position.y = 0.06;
  group.add(alertRing);

  return group;
}

    const tentRoof = new THREE.Mesh(new THREE.ConeGeometry(2.4, 1.25, 6), new THREE.MeshStandardMaterial({ color: 0x354025 }));
    tentRoof.position.y = 2.75; tentRoof.castShadow = true;
    const crate = new THREE.Mesh(new THREE.BoxGeometry(0.95, 0.68, 0.75), woodMat);
    crate.position.set(2.15, 0.34, 1.2); crate.castShadow = true;
    group.add(tentBody, tentRoof, crate);
  }

  const targetRing = new THREE.Mesh(
    new THREE.RingGeometry(BUILDINGS_DB[b.type].radius * 0.92, BUILDINGS_DB[b.type].radius * 1.12, 36),
    new THREE.MeshBasicMaterial({ color: 0xff2d20, side: THREE.DoubleSide, transparent: true, opacity: 0.9 })
  );
  targetRing.rotation.x = -Math.PI / 2;
  targetRing.position.y = 0.09;
  targetRing.visible = false;
  group.add(targetRing);
  group.userData.targetRing = targetRing;

  if (showHpBar) {
    const hpSprite = createHealthBarSprite(3.6, 0.46);
    hpSprite.position.set(0, topHeight, 0);
    group.add(hpSprite);
    group.userData.hpSprite = hpSprite;
  }

  return group;
}

function createRealisticSicario3D(shirtColor, hatColor, weaponType = "pistolero", markerColor = null, showHpBar = false) {
  const root = new THREE.Group();
  const skinMat = new THREE.MeshStandardMaterial({ color: 0xd8a273, roughness: 0.7 });
  const shirtMat = new THREE.MeshStandardMaterial({ color: shirtColor, roughness: 0.85 });
  const pantsMat = new THREE.MeshStandardMaterial({ color: 0x22262b, roughness: 0.9 });
  const gunMat = new THREE.MeshStandardMaterial({ color: 0x181a1c, metalness: 0.85, roughness: 0.25 });

  const leftLeg = new THREE.Group(); leftLeg.position.set(-0.16, 0.85, 0);
  const lThigh = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.09, 0.85, 8), pantsMat);
  lThigh.position.y = -0.42; lThigh.castShadow = true; leftLeg.add(lThigh);

  const rightLeg = new THREE.Group(); rightLeg.position.set(0.16, 0.85, 0);
  const rThigh = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.09, 0.85, 8), pantsMat);
  rThigh.position.y = -0.42; rThigh.castShadow = true; rightLeg.add(rThigh);

  const torso = new THREE.Mesh(new THREE.BoxGeometry(0.54, 0.78, 0.3), shirtMat);
  torso.position.y = 1.24; torso.castShadow = true;

  const vestThick = weaponType === "metralleta" ? 0.39 : 0.34;
  const vest = new THREE.Mesh(new THREE.BoxGeometry(0.58, 0.58, vestThick), new THREE.MeshStandardMaterial({ color: 0x161915, roughness: 0.75 }));
  vest.position.y = 1.28;

  const head = new THREE.Mesh(new THREE.SphereGeometry(0.21, 12, 12), skinMat);
  head.position.y = 1.84; head.castShadow = true;
  const hatBrim = new THREE.Mesh(new THREE.CylinderGeometry(0.34, 0.34, 0.04, 14), new THREE.MeshStandardMaterial({ color: hatColor }));
  hatBrim.position.y = 1.98;
  const hatTop = new THREE.Mesh(new THREE.CylinderGeometry(0.19, 0.21, 0.16, 12), new THREE.MeshStandardMaterial({ color: hatColor }));
  hatTop.position.y = 2.06;

  const armsGroup = new THREE.Group(); armsGroup.position.set(0, 1.48, 0.1);
  const rightArm = new THREE.Mesh(new THREE.CylinderGeometry(0.075, 0.065, 0.62, 8), skinMat);
  rightArm.rotation.x = Math.PI / 2.2; rightArm.position.set(0.32, -0.05, 0.25);
  const leftArm = new THREE.Mesh(new THREE.CylinderGeometry(0.075, 0.065, 0.62, 8), skinMat);
  leftArm.rotation.x = Math.PI / 2.4; leftArm.rotation.z = -0.35; leftArm.position.set(-0.22, -0.08, 0.28);

  if (weaponType === "granadero") {
    const launcher = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.09, 0.95, 10), new THREE.MeshStandardMaterial({ color: 0x3b4426, metalness: 0.6 }));
    launcher.rotation.x = Math.PI / 2; launcher.position.set(0.26, -0.02, 0.62);
    armsGroup.add(launcher);
  } else if (weaponType === "metralleta") {
    const receiver = new THREE.Mesh(new THREE.BoxGeometry(0.09, 0.16, 0.72), gunMat);
    receiver.position.set(0.26, -0.06, 0.5);
    const mag = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.26, 0.1), gunMat);
    mag.position.set(0.26, -0.22, 0.55); mag.rotation.x = 0.25;
    const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.025, 0.025, 0.5, 6), gunMat);
    barrel.rotation.x = Math.PI / 2; barrel.position.set(0.26, -0.03, 0.98);
    armsGroup.add(receiver, mag, barrel);
  } else {
    const receiver = new THREE.Mesh(new THREE.BoxGeometry(0.07, 0.12, 0.45), gunMat);
    receiver.position.set(0.26, -0.06, 0.42);
    armsGroup.add(receiver);
  }
  armsGroup.add(rightArm, leftArm);

  if (markerColor !== null) {
    const marker = new THREE.Mesh(
      new THREE.RingGeometry(0.38, 0.48, 20),

    hpSprite = createHealthBarSprite(1.55, 0.24);
    hpSprite.position.set(0, 2.55, 0);
    root.add(hpSprite);
  }

  root.add(leftLeg, rightLeg, torso, vest, head, hatBrim, hatTop, armsGroup);
  root.userData = { leftLeg, rightLeg, hpSprite, walkPhase: Math.random() * Math.PI * 2, steerSign: Math.random() > 0.5 ? 1 : -1 };
  return root;
}

function animateCharacterWalk(mesh, isMoving) {
  if (!mesh || !mesh.userData) return;
  if (isMoving) {
    mesh.userData.walkPhase += 0.13;
    const swing = Math.sin(mesh.userData.walkPhase) * 0.48;
    mesh.userData.leftLeg.rotation.x = swing;
    mesh.userData.rightLeg.rotation.x = -swing;
  } else {
    mesh.userData.leftLeg.rotation.x = 0;
    mesh.userData.rightLeg.rotation.x = 0;
  }
}

function moveWithObstacleAvoidance(entity, targetX, targetZ, speed, buildingsList) {
  let dx = targetX - entity.x;
  let dz = targetZ - entity.z;
  const dist = Math.hypot(dx, dz);
  if (dist < 0.05) return;

  let dirX = dx / dist;
  let dirZ = dz / dist;

  const lookAhead = 1.8;
  const steerSign = (entity.mesh && entity.mesh.userData && entity.mesh.userData.steerSign) || 1;

  buildingsList.forEach(b => {
    if ((b.hp !== undefined && b.hp <= 0) || b.destroyed) return;
    const safeRadius = BUILDINGS_DB[b.type].radius + lookAhead;
    const bx = entity.x - b.x;
    const bz = entity.z - b.z;
    const bDist = Math.hypot(bx, bz);

    if (bDist < safeRadius) {
      const nx = bDist > 0.001 ? bx / bDist : 1;
      const nz = bDist > 0.001 ? bz / bDist : 0;
      const tx = -nz * steerSign;
      const tz = nx * steerSign;
      const weight = (safeRadius - bDist) / safeRadius;

      dirX += (tx * 1.45 + nx * 0.95) * weight;
      dirZ += (tz * 1.45 + nz * 0.95) * weight;
    }
  });

  const finalLen = Math.hypot(dirX, dirZ) || 1;
  dirX /= finalLen;
  dirZ /= finalLen;

  entity.x += dirX * speed;
  entity.z += dirZ * speed;

  buildingsList.forEach(b => {
    if ((b.hp !== undefined && b.hp <= 0) || b.destroyed) return;
    const minR = BUILDINGS_DB[b.type].radius + 0.65;
    const odx = entity.x - b.x;
    const odz = entity.z - b.z;
    const odist = Math.hypot(odx, odz);
    if (odist < minR) {
      const onx = odist > 0.001 ? odx / odist : 1;
      const onz = odist > 0.001 ? odz / odist : 0;
      entity.x = b.x + onx * (minR + 0.05);
      entity.z = b.z + onz * (minR + 0.05);
    }
  });

  if (entity.mesh) {
    entity.mesh.position.set(entity.x, 0, entity.z);
    entity.mesh.rotation.y = Math.atan2(dirX, dirZ);
  }
}

let mode = "BASE";

function pickValidWaypoint() {
  for (let i = 0; i < 15; i++) {
    const wx = (Math.random() - 0.5) * 30;
    const wz = (Math.random() - 0.5) * 30;
    const blocked = myBuildings.some(b => !b.destroyed && Math.hypot(b.x - wx, b.z - wz) < BUILDINGS_DB[b.type].radius + 1.4);
    if (!blocked) return { x: wx, z: wz };
  }
  return { x: 0, z: 14 };
}

function rebuildBase3D() {
  while (worldGroup.children.length) worldGroup.remove(worldGroup.children[0]);
  myBuildings.forEach(b => {
    const isUnderPoliceAttack = (currentEvent && currentEvent.type === "POLICE" && !b.destroyed);
    const mesh = createBuildingMesh3D(b, isUnderPoliceAttack);
    b.mesh = mesh;
    if (isUnderPoliceAttack && mesh.userData.hpSprite) {
      updateHealthBarSprite(mesh.userData.hpSprite, b.hp, getBuildingMaxHp(b), false);
    }
    worldGroup.add(mesh);
  });

  baseDefenders = [];
  Object.keys(player.troops).forEach(k => {
    const spec = TROOPS_DB[k];
    for (let i = 0; i < player.troops[k]; i++) {
      const mesh = createRealisticSicario3D(spec.shirt, spec.hat, k, null, true);
      if (mesh.userData.hpSprite) mesh.userData.hpSprite.visible = false;
      const start = pickValidWaypoint();
      const wp = pickValidWaypoint();
      mesh.position.set(start.x, 0, start.z);
      worldGroup.add(mesh);
      baseDefenders.push({
        type: k, mesh, x: start.x, z: start.z,
        tx: wp.x, tz: wp.z,
        hp: spec.hp, maxHp: spec.hp, defense: spec.defense,
        speed: spec.speed * 0.65, cd: 0, dmg: spec.dmg, fireRate: spec.fireRate, range: spec.range
      });
    }
  });
  policeRaiders.forEach(p => { if (p.hp > 0) worldGroup.add(p.mesh); });
}
rebuildBase3D();

function triggerRandomEvent(forcedType = null) {
  if (mode !== "BASE" || currentEvent) return;
  const evType = forcedType || (Math.random() < 0.5 ? "FIRE" : "POLICE");

  if (evType === "FIRE") {
    const farms = myBuildings.filter(b => b.type === "plantacion" && !b.destroyed);
    if (farms.length === 0) return;
    const targetFarm = farms[Math.floor(Math.random() * farms.length)];
    targetFarm.burning = true;
    currentEvent = { type: "FIRE", buildingId: targetFarm.id, deadline: Date.now() + 28000 };
    rebuildBase3D();
    SoundFX.play("siren");

    const banner = document.getElementById("event-banner");
    document.getElementById("event-banner-txt").innerHTML = `${ICONS.fire} ¡INCENDIO EN PLANTACIÓN! (28s)`;
    const btn = document.getElementById("event-banner-btn");
    btn.style.display = "inline-flex"; btn.textContent = "Apagar ($120)";
    banner.style.display = "flex";
    showToast("¡ALERTA! Apaga el incendio antes de que se destruya");
  } else if (evType === "POLICE") {
    currentEvent = { type: "POLICE" };
    policeRaiders = [];
    myBuildings.forEach(b => {
      if (!b.destroyed) b.hp = getBuildingMaxHp(b);
    });
    rebuildBase3D();

    baseDefenders.forEach(d => {
      d.hp = d.maxHp;
      if (d.mesh.userData.hpSprite) {
        d.mesh.userData.hpSprite.visible = true;
        updateHealthBarSprite(d.mesh.userData.hpSprite, d.hp, d.maxHp);
      }
    });

    const copCount = 3 + Math.min(4, Math.floor(player.level / 2));
    for (let i = 0; i < copCount; i++) {
      const mesh = createRealisticSicario3D(0x1b365d, 0x0f1f38, "metralleta", 0x3498db, true);
      const px = -8 + i * 4, pz = 26;
      const copHp = 135 + player.level * 15;
      mesh.position.set(px, 0, pz);
      worldGroup.add(mesh);
      policeRaiders.push({ mesh, x: px, z: pz, hp: copHp, maxHp: copHp, dmg: 22, range: 6.5, speed: 0.032, cd: 0 });
    }
    SoundFX.play("siren");
    const banner = document.getElementById("event-banner");
    document.getElementById("event-banner-txt").innerHTML = `${ICONS.police} ¡REDADA POLICIAL! Defiende tus edificios`;
    document.getElementById("event-banner-btn").style.display = "none";
    banner.style.display = "flex";
    showToast("¡LA POLICÍA ESTÁ ASALTANDO Y DEMOLIENDO TU CAMPAMENTO!");
  }
}

function resolveActiveEvent() {
  if (!currentEvent || currentEvent.type !== "FIRE") return;

  currentEvent = null;

  rebuildBase3D(); updateHUD(); saveGameProgress();
  SoundFX.play("build");
  showToast("¡Incendio sofocado! Plantación a salvo");
}

// ================= COLOCACIÓN, MEJORA Y RECONSTRUCCIÓN DE EDIFICACIONES =================
let placingState = null;

function isPositionValid(x, z, radius) {
  if (Math.hypot(x, z) > 33) return false;
  return !myBuildings.some(b => Math.hypot(b.x - x, b.z - z) < (radius + BUILDINGS_DB[b.type].radius + 0.9));
}

function enterPlacementMode(type) {
  SoundFX.play("click");
  const spec = BUILDINGS_DB[type];
  if (spec.costCash && player.cash < spec.costCash) return showToast(`Necesitas $${spec.costCash}`);
  if (spec.costGoods && player.goods < spec.costGoods) return showToast(`Necesitas ${spec.costGoods} de Mercancía`);

  closeModals(); cancelPlacement();
  const ghost = createBuildingMesh3D({ type, x: 0, z: 0 }, false);
  ghost.traverse(c => { if (c.isMesh) { c.material = c.material.clone(); c.material.transparent = true; c.material.opacity = 0.65; } });

  const ring = new THREE.Mesh(
    new THREE.RingGeometry(spec.radius * 0.85, spec.radius, 32),
    new THREE.MeshBasicMaterial({ color: 0x2ecc71, side: THREE.DoubleSide })
  );
  ring.rotation.x = -Math.PI / 2; ring.position.y = 0.08; ghost.add(ring);

  placingState = { type, ghostGroup: ghost, ringMesh: ring, x: camTarget.x, z: camTarget.z, valid: true };
  updateGhostPosition(camTarget.x, camTarget.z);
  scene.add(ghost);

  document.getElementById("main-bottom-bar").style.display = "none";
  document.getElementById("place-hud").style.display = "block";
  document.getElementById("place-label").textContent = `Colocando: ${spec.name} (Arrastra o toca el suelo)`;
}

function updateGhostPosition(x, z) {
  if (!placingState) return;
  placingState.x = x; placingState.z = z;
  placingState.ghostGroup.position.set(x, 0, z);
  placingState.valid = isPositionValid(x, z, BUILDINGS_DB[placingState.type].radius);
  placingState.ringMesh.material.color.setHex(placingState.valid ? 0x2ecc71 : 0xe74c3c);
}

function cancelPlacement() {
  if (placingState) { scene.remove(placingState.ghostGroup); placingState = null; }
  document.getElementById("place-hud").style.display = "none";
  document.getElementById("main-bottom-bar").style.display = "flex";
}

function confirmPlacement() {
  if (!placingState || !placingState.valid) return showToast("Espacio bloqueado o fuera del valle");
  const { type, x, z } = placingState;
  const spec = BUILDINGS_DB[type];
  if (spec.costCash) player.cash -= spec.costCash;
  if (spec.costGoods) player.goods -= spec.costGoods;
  updateHUD(); cancelPlacement();

  const timerObj = {
    id: Date.now(),
    label: `Construyendo ${spec.name}`,
    endTime: Date.now() + spec.buildSec * 1000,
    actionType: "build",
    payload: { type, x, z }
  };
  timerObj.onComplete = hydrateTimerCallback(timerObj);
  activeTimers.push(timerObj);
  saveGameProgress();
  SoundFX.play("build");
  showToast(`Construyendo ${spec.name} (${spec.buildSec}s)...`);
}

function rebuildDestroyedBuilding(bId) {
  const b = myBuildings.find(item => item.id === bId);
  if (!b || !b.destroyed) return;
  const spec = BUILDINGS_DB[b.type];
  const repairCost = getBuildingRepairCost(b);
  if (player.cash < repairCost) return showToast(`Necesitas $${repairCost} para reconstruir ${spec.name}`);

  player.cash -= repairCost;
  b.destroyed = false;
  b.burning = false;
  b.hp = getBuildingMaxHp(b);
  rebuildBase3D();
  updateHUD();
  saveGameProgress();
  SoundFX.play("build");
  showToast(`¡${spec.name} reconstruida por $${repairCost}!`);
  openModal("upgrade");
}

function rebuildAllDestroyed() {
  const ruined = myBuildings.filter(b => b.destroyed);
  if (ruined.length === 0) return;
  const totalCost = ruined.reduce((acc, b) => acc + getBuildingRepairCost(b), 0);
  if (player.cash < totalCost) return showToast(`Necesitas $${totalCost} para reconstruir todo`);

  player.cash -= totalCost;
  ruined.forEach(b => {
    b.destroyed = false;
    b.burning = false;
    b.hp = getBuildingMaxHp(b);
  });
  rebuildBase3D();
  updateHUD();
  saveGameProgress();
  SoundFX.play("build");
  showToast(`¡Todas las edificaciones reconstruidas por $${totalCost}!`);
  openModal("upgrade");
}

    payload: { tKey }

  activeTimers.push(timerObj);

}

function finishTimerNow(id) {
  if (player.gold < 3) return showToast("Necesitas 3 de Oro");
  const idx = activeTimers.findIndex(t => t.id === id);
  if (idx === -1) return;
  player.gold -= 3;
  activeTimers.splice(idx, 1)[0].onComplete();
  updateHUD(); saveGameProgress();
}

  const box = document.getElementById("timers-box");

  }).join("");
}

function collectAll() {

  myBuildings.forEach(b => {
    if (b.destroyed) return;
    const s = BUILDINGS_DB[b.type];
    if (s.prod && b.stored >= 1) {
      const amt = Math.floor(b.stored);
      if (s.prod === "cash") { player.cash += amt; c += amt; }
      else { player.goods += amt; g += amt; }
      b.stored = 0;
    }
  });
  if (c || g) {
    SoundFX.play("cash");
    addXP(20);
    showToast(`Recogido: +$${c} Efectivo y +${g} Mercancía`);
  } else {
    SoundFX.play("click");
    const hasRuins = myBuildings.some(b => b.destroyed);
    showToast(hasRuins ? "Tienes edificaciones en ruinas sin producir" : "Aún no hay suficientes recursos acumulados");
  }
  updateHUD(); saveGameProgress();
}

function attackSelectedRival(rivalId) {

  activeRivalTarget = rival;

  cancelPlacement(); mode = "RAID"; closeModals();

  document.getElementById("raid-hud").style.display = "block";

    const fallbackHp = BUILDINGS_DB[b.type] ? BUILDINGS_DB[b.type].baseHp : 500;
    const realHp = b.hp || fallbackHp;

const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();

function getGroundHit(clientX, clientY) {
  pointer.x = (clientX / window.innerWidth) * 2 - 1;
  pointer.y = -(clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(pointer, camera);
  const hits = raycaster.intersectObject(ground);
  return hits.length > 0 ? hits[0].point : null;
}

function handleWorldTap(clientX, clientY) {
  closeModals();
  const pt = getGroundHit(clientX, clientY);
  if (!pt) return;

  if (mode === "BASE" && placingState) { updateGhostPosition(pt.x, pt.z); return; }

  if (mode === "BASE" && !placingState) {
    const tappedRuins = myBuildings.find(b => b.destroyed && Math.hypot(pt.x - b.x, pt.z - b.z) <= BUILDINGS_DB[b.type].radius + 0.8);
    if (tappedRuins) {
      openModal("upgrade");
      showToast(`${BUILDINGS_DB[tappedRuins.type].name} en ruinas: toca Reconstruir`);
      return;
    }
  }

  if (mode === "RAID") {
    const tappedBuilding = rivalBuildings.find(b => b.hp > 0 && Math.hypot(pt.x - b.x, pt.z - b.z) <= BUILDINGS_DB[b.type].radius + 0.9);
    if (tappedBuilding) {
      setPriorityTargetBuilding(tappedBuilding);
      return;
    }

    if (pt.z < 7) return showToast("Despliega tropas al sur o toca un edificio para atacarlo");
    if ((raidStock[selectedTroop] || 0) <= 0) return showToast("No quedan sicarios de este tipo");

    raidStock[selectedTroop]--; player.troops[selectedTroop]--;
    const spec = TROOPS_DB[selectedTroop];
    const mesh = createRealisticSicario3D(spec.shirt, spec.hat, selectedTroop, 0x2ecc71, true);
    mesh.position.set(pt.x, 0, pt.z); worldGroup.add(mesh);
    invadingTroops.push({
      type: selectedTroop, mesh, x: pt.x, z: pt.z,
      hp: spec.hp, maxHp: spec.hp, defense: spec.defense, cd: 0
    });
    SoundFX.play("click");
    renderRaidDock();
    saveGameProgress();
  }
}

function updateRaidLogic() {
  if (selectedTargetBuilding && selectedTargetBuilding.hp <= 0) {
    selectedTargetBuilding = null;
  }

  invadingTroops.forEach(t => {
    if (t.hp <= 0) return;
    const spec = TROOPS_DB[t.type];
    let target = null, minDist = 999, targetIsBldg = false;

    if (selectedTargetBuilding && selectedTargetBuilding.hp > 0) {
      target = selectedTargetBuilding;
      minDist = Math.hypot(target.x - t.x, target.z - t.z);
      targetIsBldg = true;
    } else {
      rivalDefenders.forEach(d => {
        if (d.hp <= 0) return;
        const dist = Math.hypot(d.x - t.x, d.z - t.z);
        if (dist < minDist) { minDist = dist; target = d; targetIsBldg = false; }
      });
      if (!target) {
        rivalBuildings.forEach(b => {
          if (b.hp <= 0) return;
          const dist = Math.hypot(b.x - t.x, b.z - t.z);
          if (dist < minDist) { minDist = dist; target = b; targetIsBldg = true; }
        });
      }
    }

    if (!target) return;

    const stopDist = targetIsBldg ? (spec.range + BUILDINGS_DB[target.type].radius * 0.7) : spec.range;
    if (minDist > stopDist) {
      moveWithObstacleAvoidance(t, target.x, target.z, spec.speed, rivalBuildings);
      animateCharacterWalk(t.mesh, true);
    } else {
      t.mesh.rotation.y = Math.atan2(target.x - t.x, target.z - t.z);
      animateCharacterWalk(t.mesh, false);
      t.cd--;
      if (t.cd <= 0) {
        t.cd = spec.fireRate;
        const finalDmg = targetIsBldg ? Math.floor(spec.dmg * spec.bldgBonus) : Math.floor(spec.dmg * (1 - (target.defense || 0)));
        target.hp -= finalDmg;

        spawnBulletTracer(t.x, t.z, target.x, target.z, t.type === "granadero" ? 0xff7675 : 0xffeaa7);
        SoundFX.play(t.type === "granadero" ? "explosion" : (t.type === "metralleta" ? "rifle" : "pistol"));

        if (targetIsBldg && target.mesh.userData.hpSprite) {
          updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, target.maxHp, target === selectedTargetBuilding);
        } else if (!targetIsBldg && target.mesh.userData.hpSprite) {
          updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, target.maxHp);
        }

        if (target.hp <= 0) {
          target.mesh.visible = false;
          if (targetIsBldg) SoundFX.play("explosion");
        }
      }
    }
  });

  rivalDefenders.forEach(d => {
    if (d.hp <= 0) return;
    let target = null, minDist = 999;
    invadingTroops.forEach(t => {
      if (t.hp <= 0) return;
      const dist = Math.hypot(t.x - d.x, t.z - d.z);
      if (dist < minDist) { minDist = dist; target = t; }
    });
    if (!target) { animateCharacterWalk(d.mesh, false); return; }

    if (minDist > d.range) {
      moveWithObstacleAvoidance(d, target.x, target.z, d.speed, rivalBuildings);
      animateCharacterWalk(d.mesh, true);
    } else {
      d.mesh.rotation.y = Math.atan2(target.x - d.x, target.z - d.z);
      animateCharacterWalk(d.mesh, false);
      d.cd--;
      if (d.cd <= 0) {
        d.cd = d.fireRate;
        const mitigatedDmg = Math.max(2, Math.floor(d.dmg * (1 - (target.defense || 0))));
        target.hp -= mitigatedDmg;
        spawnBulletTracer(d.x, d.z, target.x, target.z, 0xff7675);
        if (target.mesh.userData.hpSprite) {
          updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, target.maxHp);
        }
        if (target.hp <= 0) target.mesh.visible = false;
      }
    }
  });

  rivalBuildings.forEach(b => {
    if (b.hp <= 0 || b.type !== "torreta") return;
    b.cd = (b.cd || 0) - 1;
    if (b.cd <= 0) {
      const target = invadingTroops.find(t => t.hp > 0 && Math.hypot(t.x - b.x, t.z - b.z) <= 10.5);
      if (target) {
        b.cd = 36;
        const mitigatedDmg = Math.max(3, Math.floor(22 * (1 - (target.defense || 0))));
        target.hp -= mitigatedDmg;
        spawnBulletTracer(b.x, b.z, target.x, target.z, 0xff4757);
        SoundFX.play("rifle");
        if (target.mesh.userData.hpSprite) {
          updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, target.maxHp);
        }
        if (target.hp <= 0) target.mesh.visible = false;
      }
    }
  });

  const finca = rivalBuildings.find(b => b.type === "finca");
  if (finca && finca.hp <= 0) endRaid(true);
}

function endRaid(victory) {
  mode = "BASE";
  selectedTargetBuilding = null;
  document.getElementById("raid-hud").style.display = "none";
  document.getElementById("base-hud").style.display = "block";
  invadingTroops.forEach(t => { if (t.hp > 0) player.troops[t.type]++; });
  invadingTroops = []; rebuildBase3D();

  if (victory && activeRivalTarget) {
    const stolenCash = Math.floor(activeRivalTarget.lootCash * ADMIN_CONFIG.economyMultiplier);
    const stolenGoods = Math.floor(activeRivalTarget.lootGoods * ADMIN_CONFIG.economyMultiplier);
    player.cash += stolenCash; player.goods += stolenGoods; addXP(280);
    SoundFX.play("cash");
    showToast(`¡SAQUEO EXITOSO! Robaste $${stolenCash} y ${stolenGoods} Mercancía`);
  } else {
    SoundFX.play("click");
    showToast("Regresaste a tu Hacienda");
  }
  updateHUD();
  saveGameProgress();
}

// ================= CONTROLES TÁCTILES =================
let lastX = 0, lastY = 0, moved = false, pinchStart = 0, draggingBuilding = false;
renderer.domElement.addEventListener("touchstart", e => {
  if (e.touches.length === 1) {
    lastX = e.touches[0].clientX; lastY = e.touches[0].clientY; moved = false; draggingBuilding = false;
    if (placingState) {
      const pt = getGroundHit(lastX, lastY);
      if (pt && Math.hypot(pt.x - placingState.x, pt.z - placingState.z) < 5.5) draggingBuilding = true;
    }
  } else if (e.touches.length === 2) {
    pinchStart = Math.hypot(e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY);
  }
}, { passive: false });

renderer.domElement.addEventListener("touchmove", e => {
  e.preventDefault();
  if (e.touches.length === 1) {
    const cx = e.touches[0].clientX, cy = e.touches[0].clientY;
    const dx = cx - lastX, dy = cy - lastY;
    if (Math.hypot(dx, dy) > 6) moved = true;
    if (draggingBuilding && placingState) {
      const pt = getGroundHit(cx, cy);
      if (pt) updateGhostPosition(pt.x, pt.z);
    } else if (moved) {
      camTarget.x = Math.max(-45, Math.min(45, camTarget.x - dx * 0.05));
      camTarget.z = Math.max(-45, Math.min(45, camTarget.z - dy * 0.05));
      lastX = cx; lastY = cy;
    }
  } else if (e.touches.length === 2) {
    const d = Math.hypot(e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY);
    if (pinchStart > 0) camDist = Math.max(16, Math.min(58, camDist - (d - pinchStart) * 0.06));
    pinchStart = d;
  }
}, { passive: false });

renderer.domElement.addEventListener("touchend", e => {
  if (!moved && e.changedTouches.length === 1) handleWorldTap(e.changedTouches[0].clientX, e.changedTouches[0].clientY);
  draggingBuilding = false; pinchStart = 0;
});
renderer.domElement.addEventListener("click", e => handleWorldTap(e.clientX, e.clientY));

function sendTargetedEventToUser() {
  const targetUserId = document.getElementById("adm-target-user").value;
  const evType = document.getElementById("adm-target-event").value;

  if (targetUserId === player.id) {
    closeModals();
    triggerRandomEvent(evType);
    return;
  }

  const targetObj = onlinePlayersDirectory.find(u => u.id === targetUserId);
  if (targetObj) {
    targetObj.pendingEvent = evType;
    if (ADMIN_CONFIG.firebaseDbUrl) {
      fetch(`${ADMIN_CONFIG.firebaseDbUrl}/players/${targetUserId}/pendingEvent.json`, {
        method: "PUT", body: JSON.stringify(evType)
      }).catch(() => {});
    }
    showToast(`Evento (${evType}) enviado a ${targetObj.name}`);
  }
}

function updateHUD() {

  document.getElementById("ui-rank").textContent = getCurrentRankName();
  document.getElementById("ui-xp-bar").style.width = `${Math.min(100, (player.xp / req) * 100)}%`;
  document.getElementById("ui-squad").textContent = `Sicarios: ${total}/${getMaxTroops()} • Rangos ▸`;
  document.getElementById("ui-cash").textContent = player.cash.toLocaleString();
  document.getElementById("ui-goods").textContent = player.goods.toLocaleString();
  document.getElementById("ui-gold").textContent = player.gold.toLocaleString();

  const ruinedCount = myBuildings.filter(b => b.destroyed).length;
  const badge = document.getElementById("badge-ruins");
  if (ruinedCount > 0) {
    badge.style.display = "flex";
    badge.textContent = ruinedCount;
  } else {
    badge.style.display = "none";
  }
}

function showToast(msg) {
  const t = document.getElementById("toast");
  t.textContent = msg; t.style.opacity = 1;
  clearTimeout(window.toastT);
  window.toastT = setTimeout(() => t.style.opacity = 0, 2600);
}

function animate() {
  requestAnimationFrame(animate);
  camera.position.set(camTarget.x, camDist * 0.68, camTarget.z + camDist * 0.85);
  camera.lookAt(camTarget);

  if (mode === "BASE") {
    const activeCops = policeRaiders.filter(p => p.hp > 0);
    baseDefenders.forEach(d => {
      if (d.hp <= 0) return;
      if (activeCops.length > 0) {
        let nearestCop = activeCops[0], minDist = 999;
        activeCops.forEach(c => { const dist = Math.hypot(c.x - d.x, c.z - d.z); if (dist < minDist) { minDist = dist; nearestCop = c; } });
        if (minDist > d.range) {
          moveWithObstacleAvoidance(d, nearestCop.x, nearestCop.z, d.speed, myBuildings);
          animateCharacterWalk(d.mesh, true);
        } else {
          d.mesh.rotation.y = Math.atan2(nearestCop.x - d.x, nearestCop.z - d.z);
          animateCharacterWalk(d.mesh, false);
          d.cd--;
          if (d.cd <= 0) {
            d.cd = d.fireRate; nearestCop.hp -= d.dmg;
            spawnBulletTracer(d.x, d.z, nearestCop.x, nearestCop.z, 0xffeaa7);
            SoundFX.play(d.type === "granadero" ? "explosion" : "pistol");
            if (nearestCop.mesh.userData.hpSprite) {
              updateHealthBarSprite(nearestCop.mesh.userData.hpSprite, nearestCop.hp, nearestCop.maxHp);
            }
            if (nearestCop.hp <= 0) nearestCop.mesh.visible = false;
          }
        }
      } else {
        if (d.mesh.userData.hpSprite && d.mesh.userData.hpSprite.visible) {
          d.mesh.userData.hpSprite.visible = false;
        }
        const distToWp = Math.hypot(d.tx - d.x, d.tz - d.z);
        if (distToWp < 1.3) {
          const nextWp = pickValidWaypoint();
          d.tx = nextWp.x; d.tz = nextWp.z;
        } else {
          moveWithObstacleAvoidance(d, d.tx, d.tz, d.speed, myBuildings);
          animateCharacterWalk(d.mesh, true);
        }
      }
    });

    if (currentEvent && currentEvent.type === "POLICE") {
      const aliveCops = policeRaiders.filter(p => p.hp > 0);
      if (aliveCops.length === 0) {
        currentEvent = null;
        document.getElementById("event-banner").style.display = "none";
        rebuildBase3D();
        addXP(220); showToast("¡Redada policial repelida con éxito! (+XP)");
      } else {
        aliveCops.forEach(cop => {
          const aliveDefs = baseDefenders.filter(def => def.hp > 15);
          const intactBldgs = myBuildings.filter(b => !b.destroyed && b.hp > 0);

          let target = null, minDist = 999, isDef = false;
          aliveDefs.forEach(def => {
            const dist = Math.hypot(def.x - cop.x, def.z - cop.z);
            if (dist < minDist) { minDist = dist; target = def; isDef = true; }
          });

          if (!target && intactBldgs.length > 0) {
            intactBldgs.forEach(b => {
              const dist = Math.hypot(b.x - cop.x, b.z - cop.z);
              if (dist < minDist) { minDist = dist; target = b; isDef = false; }
            });
          }

          if (!target) return;

          const stopDist = isDef ? cop.range : (cop.range + BUILDINGS_DB[target.type].radius * 0.65);
          if (minDist > stopDist) {
            moveWithObstacleAvoidance(cop, target.x, target.z, cop.speed, myBuildings);
            animateCharacterWalk(cop.mesh, true);
          } else {
            cop.mesh.rotation.y = Math.atan2(target.x - cop.x, target.z - cop.z);
            animateCharacterWalk(cop.mesh, false);
            cop.cd--;
            if (cop.cd <= 0) {
              cop.cd = 38;
              spawnBulletTracer(cop.x, cop.z, target.x, target.z, 0x3498db);
              if (isDef) {
                const dmgTaken = Math.max(2, Math.floor(cop.dmg * (1 - (target.defense || 0))));
                target.hp = Math.max(15, target.hp - dmgTaken);
                if (target.mesh.userData.hpSprite) {
                  updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, target.maxHp);
                }
              } else {
                target.hp -= cop.dmg * 1.4;
                if (target.mesh && target.mesh.userData.hpSprite) {
                  updateHealthBarSprite(target.mesh.userData.hpSprite, target.hp, getBuildingMaxHp(target));
                }
                if (target.hp <= 0) {
                  destroyBaseBuilding(target);
                }
              }
            }
          }
        });
      }
    }
  } else {
    updateRaidLogic();
  }

  for (let i = activeTracers.length - 1; i >= 0; i--) {
    activeTracers[i].life--;
    if (activeTracers[i].life <= 0) {
      worldGroup.remove(activeTracers[i].line);
      activeTracers.splice(i, 1);
    }
  }
  renderer.render(scene, camera);
}

if (!loadSavedGame()) {
  updateHUD();
}
animate();
